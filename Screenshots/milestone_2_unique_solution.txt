```python
def count_solutions(board, limit=2):
    count = [0]
    def backtrack():
        if count[0] >= limit:
            return
        for row in range(9):
            for col in range(9):
                if board[row][col] == 0:
                    for num in range(1, 10):
                        if is_safe(board, row, col, num):
                            board[row][col] = num
                            backtrack()
                            board[row][col] = 0
                            if count[0] >= limit:
                                return
                    return
        count[0] += 1
    backtrack()
    return count[0]
```
